import { Injectable, ForbiddenException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { NotificationsService } from '../notifications/notifications.service';

@Injectable()
export class AnnouncementsService {
  constructor(
    private prisma: PrismaService,
    private notifications: NotificationsService
  ) {}

  async create(data: { title: string, content: string, organizationId: string }, userId: string) {
    // 1. Verify user is a Club Head
    const membership = await this.prisma.membership.findUnique({
      where: { userId_organizationId: { userId, organizationId: data.organizationId } },
    });

    if (!membership || (membership.role !== 'CLUB_HEAD' && membership.role !== 'COMMITTEE_MEMBER')) {
      throw new ForbiddenException('Only Club Heads can post announcements');
    }

    // 2. Create the announcement
    const announcement = await this.prisma.announcement.create({
      data: {
        title: data.title,
        content: data.content,
        organizationId: data.organizationId,
      },
    });

    // 3. Find all members of the club to notify
    const members = await this.prisma.membership.findMany({
      where: { organizationId: data.organizationId, isActive: true },
    });

    // 4. Send notifications to everyone (except the sender)
    const org = await this.prisma.organization.findUnique({ where: { id: data.organizationId } });
    const orgName = org?.name || 'Club'; // FIX: Added fallback for TypeScript

    const promises = members
      .filter(m => m.userId !== userId) // Don't notify the sender
      .map(m => this.notifications.create(m.userId, `?? ${orgName} Announcement`, data.title));

    await Promise.all(promises);

    return announcement;
  }

  async getClubAnnouncements(orgId: string) {
    return this.prisma.announcement.findMany({
      where: { organizationId: orgId },
      orderBy: { createdAt: 'desc' },
    });
  }
}

import { Injectable, ConflictException, UnauthorizedException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { JwtService } from '@nestjs/jwt';
import * as argon2 from 'argon2';
import { RegisterDto } from './dto/register.dto';
import { LoginDto } from './dto/login.dto';

@Injectable()
export class AuthService {
  constructor(
    private prisma: PrismaService,
    private jwtService: JwtService,
  ) {}

  async register(dto: RegisterDto) {
    const existingUser = await this.prisma.user.findUnique({ where: { email: dto.email } });
    if (existingUser) throw new ConflictException('Email already exists');

    const hash = await argon2.hash(dto.password);

    const user = await this.prisma.user.create({
      data: {
        email: dto.email,
        passwordHash: hash,
        globalRole: dto.role,
        status: 'ACTIVE',
        ...(dto.role === 'STUDENT' ? {
          studentProfile: {
            create: {
              firstName: dto.firstName, lastName: dto.lastName, rollNumber: dto.identifier,
              department: 'General', interests: [], skills: []
            }
          }
        } : dto.role === 'FACULTY' ? {
          facultyProfile: {
            create: {
              firstName: dto.firstName, lastName: dto.lastName, designation: dto.identifier, department: 'General'
            }
          }
        } : {})
      },
      include: { studentProfile: true, facultyProfile: true }
    });

    return { message: 'User registered successfully', user: { id: user.id, email: user.email, role: user.globalRole } };
  }

  async login(dto: LoginDto) {
    const user = await this.prisma.user.findUnique({ where: { email: dto.email } });
    if (!user) throw new UnauthorizedException('Invalid credentials');

    // Check if user is suspended
    if (user.status === 'SUSPENDED') {
      throw new UnauthorizedException('Your account has been suspended. Please contact the administrator.');
    }

    const isPasswordValid = await argon2.verify(user.passwordHash, dto.password);
    if (!isPasswordValid) throw new UnauthorizedException('Invalid credentials');

    const payload = { sub: user.id, email: user.email, role: user.globalRole };
    return {
      message: 'Login successful',
      access_token: this.jwtService.sign(payload),
      user: { id: user.id, email: user.email, role: user.globalRole }
    };
  }
}

import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy) {
  constructor(private prisma: PrismaService) {
    super({
      jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
      ignoreExpiration: false,
      secretOrKey: process.env.JWT_SECRET || 'super-secret-change-this-in-production',
    });
  }

  async validate(payload: any) {
    return { userId: payload.sub, email: payload.email, role: payload.role };
  }
}

import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { Reflector } from '@nestjs/core';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<string[]>('roles', [
      context.getHandler(),
      context.getClass(),
    ]);
    if (!requiredRoles) return true;
    
    const { user } = context.switchToHttp().getRequest();
    return requiredRoles.includes(user.role);
  }
}

import { Injectable, ForbiddenException, ConflictException, NotFoundException, BadRequestException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreateEventDto } from './dto/create-event.dto';

@Injectable()
export class EventsService {
  constructor(private prisma: PrismaService) {}

  async findAll(search?: string, isFest?: boolean) {
    return this.prisma.event.findMany({
      where: { 
        status: 'PUBLISHED',
        ...(search && { title: { contains: search, mode: 'insensitive' } }),
        ...(isFest !== undefined && { isFest: isFest }),
      },
      include: { organization: true, registrations: true },
      orderBy: { startDate: 'asc' },
    });
  }

  async create(dto: CreateEventDto, userId: string) {
    const membership = await this.prisma.membership.findUnique({
      where: { userId_organizationId: { userId, organizationId: dto.organizationId } },
    });
    if (!membership || (membership.role !== 'CLUB_HEAD' && membership.role !== 'COMMITTEE_MEMBER')) {
      throw new ForbiddenException('Only Club Heads can create events');
    }

    // FLAW FIX: Validate dates
    if (new Date(dto.endDate) < new Date(dto.startDate)) {
      throw new BadRequestException('End date cannot be before start date');
    }

    return this.prisma.event.create({
      data: {
        title: dto.title, description: dto.description, startDate: new Date(dto.startDate), endDate: new Date(dto.endDate),
        organizationId: dto.organizationId, status: 'PUBLISHED', isFest: dto.isFest || false, parentEventId: dto.parentEventId || null,
        maxCapacity: dto.maxCapacity || null,
      },
    });
  }

  async register(eventId: string, userId: string) {
    const user = await this.prisma.user.findUnique({ where: { id: userId } });
    
    // FLAW FIX: Only students can register
    if (user?.globalRole !== 'STUDENT') {
      throw new ForbiddenException('Only students can register for events.');
    }

    const event = await this.prisma.event.findUnique({
      where: { id: eventId },
      include: { registrations: true }
    });
    if (!event) throw new NotFoundException('Event not found');

    // FLAW FIX: Cannot register for past events
    if (new Date(event.startDate) < new Date()) {
      throw new BadRequestException('Cannot register for an event that has already started.');
    }

    const existing = await this.prisma.eventRegistration.findUnique({
      where: { userId_eventId: { userId, eventId } },
    });
    if (existing) throw new ConflictException('Already registered or waitlisted');

    let status = 'REGISTERED';
    if (event.maxCapacity) {
      const registeredCount = event.registrations.filter(r => r.status === 'REGISTERED').length;
      if (registeredCount >= event.maxCapacity) {
        status = 'WAITLISTED';
      }
    }

    return this.prisma.eventRegistration.create({
      data: { userId, eventId, status },
    });
  }

  async cancelRegistration(eventId: string, userId: string) {
    const reg = await this.prisma.eventRegistration.findUnique({
      where: { userId_eventId: { userId, eventId } }
    });
    if (!reg) throw new NotFoundException('Registration not found');

    await this.prisma.eventRegistration.delete({ where: { id: reg.id } });

    if (reg.status === 'REGISTERED') {
      const nextInLine = await this.prisma.eventRegistration.findFirst({
        where: { eventId, status: 'WAITLISTED' },
        orderBy: { id: 'asc' }
      });
      if (nextInLine) {
        await this.prisma.eventRegistration.update({
          where: { id: nextInLine.id },
          data: { status: 'REGISTERED' }
        });
      }
    }
    return { message: 'Registration cancelled successfully' };
  }

  async findMyEvents(userId: string) {
    return this.prisma.eventRegistration.findMany({
      where: { userId },
      include: { event: { include: { organization: true } } },
    });
  }

  async getEventAttendees(eventId: string) {
    return this.prisma.eventRegistration.findMany({
      where: { eventId },
      include: { user: { include: { studentProfile: true } } },
      orderBy: { user: { studentProfile: { firstName: 'asc' } } }
    });
  }

  async markAttendance(registrationId: string, status: string) {
    return this.prisma.eventRegistration.update({
      where: { id: registrationId },
      data: { status }
    });
  }
}

@WebSocketGateway({ cors: { origin: '*' } })
export class ChatGateway {
  @WebSocketServer()
  server: Server;

  constructor(private readonly messagesService: MessagesService) {}

  @SubscribeMessage('sendMessage')
  async handleMessage(
    @MessageBody() data: { senderId: string, receiverId: string, content: string },
    @ConnectedSocket() client: Socket
  ) {
    // 1. Save to Database
    const newMessage = await this.messagesService.sendMessage(data.senderId, data.receiverId, data.content);
    
    // 2. Broadcast the message to the receiver and the sender
    this.server.emit('newMessage', newMessage);
  }
}

import { Injectable, NotFoundException, BadRequestException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class MessagesService {
  constructor(private prisma: PrismaService) {}

  async getConversations(userId: string) {
    const messages = await this.prisma.message.findMany({
      where: { OR: [{ senderId: userId }, { receiverId: userId }] },
      orderBy: { createdAt: 'desc' },
      include: { sender: true, receiver: true },
    });

    const usersMap = new Map();
    messages.forEach(msg => {
      const otherUser = msg.senderId === userId ? msg.receiver : msg.sender;
      if (!usersMap.has(otherUser.id)) {
        usersMap.set(otherUser.id, {
          id: otherUser.id,
          email: otherUser.email,
          avatarUrl: otherUser.avatarUrl,
          latestMessage: msg.content,
          createdAt: msg.createdAt,
        });
      }
    });
    return Array.from(usersMap.values());
  }

  async getMessages(userId: string, otherUserId: string) {
    return this.prisma.message.findMany({
      where: {
        OR: [
          { senderId: userId, receiverId: otherUserId },
          { senderId: otherUserId, receiverId: userId },
        ],
      },
      orderBy: { createdAt: 'asc' },
    });
  }

  async sendMessage(senderId: string, receiverId: string, content: string) {
    // FLAW FIX: Cannot message yourself
    if (senderId === receiverId) {
      throw new BadRequestException('You cannot send a message to yourself.');
    }
    return this.prisma.message.create({
      data: { senderId, receiverId, content },
    });
  }
}

export class OrganizationsService {

  async findAll(search?: string, category?: string) {
    return this.prisma.organization.findMany({
      where: { 
        status: 'ACTIVE',
        ...(search && { name: { contains: search, mode: 'insensitive' } }),
        ...(category && { category: { equals: category, mode: 'insensitive' } }),

    return this.prisma.organization.findMany({
      include: { memberships: true },
      orderBy: { renewalDate: 'asc' }
    });
  }

  async findPending() {
    return this.prisma.organization.findMany({
      where: { status: 'PENDING_REVIEW' },
      include: { memberships: { include: { user: true } } }
    });
  }

  async create(dto: CreateOrganizationDto, userId: string) {
    const user = await this.prisma.user.findUnique({ where: { id: userId } });
    if (user?.globalRole !== 'STUDENT') {
      throw new ForbiddenException('Only students can create clubs and be assigned as Club Head.');
    }

    const renewalDate = new Date();
    renewalDate.setFullYear(renewalDate.getFullYear() + 1);
    return this.prisma.organization.create({
      data: {
        name: dto.name, description: dto.description, category: dto.category, status: 'PENDING_REVIEW',
        renewalDate: renewalDate,
        memberships: { create: { userId: userId, role: 'CLUB_HEAD' } },
      },
    });
  }

  async approve(id: string) {
    return this.prisma.organization.update({ where: { id }, data: { status: 'ACTIVE' } });
  }

  async renew(id: string) {
    const org = await this.prisma.organization.findUnique({ where: { id } });
    if (!org) throw new NotFoundException('Organization not found');
    const newRenewalDate = new Date(org.renewalDate);
    newRenewalDate.setFullYear(newRenewalDate.getFullYear() + 1);
    return this.prisma.organization.update({ where: { id }, data: { renewalDate: newRenewalDate } });
  }

  async join(orgId: string, userId: string) {
    const user = await this.prisma.user.findUnique({ where: { id: userId } });
    if (user?.globalRole !== 'STUDENT') {
      throw new ForbiddenException('Only students can join clubs as members.');
    }

    const existing = await this.prisma.membership.findUnique({
      where: { userId_organizationId: { userId, organizationId: orgId } },
    });
    if (existing) throw new ConflictException('Already a member');
    return this.prisma.membership.create({ data: { userId, organizationId: orgId, role: 'MEMBER' } });

      where: { facultyGuardianId: userId, status: 'ACTIVE' },
      include: { memberships: true }
    });
  }
}

import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class RecommendationsService {
  constructor(private prisma: PrismaService) {}

  async getRecommendations(userId: string) {
    // 1. Get user's skills and interests
    const profile = await this.prisma.studentProfile.findUnique({
      where: { userId },
    });

    if (!profile) return { clubs: [], events: [], opportunities: [] };

    const keywords = [...(profile.skills || []), ...(profile.interests || [])].map(k => k.toLowerCase());

    // If user has no skills/interests, return empty
    if (keywords.length === 0) return { clubs: [], events: [], opportunities: [] };

    // 2. Fetch all active data
    const [clubs, events, opportunities] = await Promise.all([
      this.prisma.organization.findMany({ where: { status: 'ACTIVE' } }),
      this.prisma.event.findMany({ where: { status: 'PUBLISHED' }, include: { organization: true } }),
      this.prisma.opportunity.findMany({ where: { status: 'PUBLISHED' } }),
    ]);

    // 3. Simple Rule-Based Matching Function
    const isMatch = (text: string) => {
      if (!text) return false;
      const lowerText = text.toLowerCase();
      return keywords.some(kw => lowerText.includes(kw));
    };

    const matchedClubs = clubs.filter(c => 
      isMatch(c.name) || isMatch(c.category) || isMatch(c.description)
    ).slice(0, 3); // Top 3

    const matchedEvents = events.filter(e => 
      isMatch(e.title) || isMatch(e.description) || (e.organization && isMatch(e.organization.name))
    ).slice(0, 3); // Top 3

    const matchedOpps = opportunities.filter(o => 
      isMatch(o.title) || isMatch(o.description) || isMatch(o.category)
    ).slice(0, 3); // Top 3

    return { clubs: matchedClubs, events: matchedEvents, opportunities: matchedOpps };
  }
}

import { Controller, Get, Post, Body, Param, UseGuards, Req, Patch, ForbiddenException, BadRequestException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { JwtAuthGuard } from '../auth/jwt-auth.guard';

@Controller('recruitment')
export class RecruitmentController {
  constructor(private prisma: PrismaService) {}

  @Get()
  getOpenCycles() {
    return this.prisma.recruitmentCycle.findMany({
      where: { status: 'OPEN' },
      include: { organization: true },
      orderBy: { createdAt: 'desc' }
    });
  }

  @UseGuards(JwtAuthGuard)
  @Post()
  async createCycle(@Req() req: any, @Body() body: any) {
    const membership = await this.prisma.membership.findUnique({
      where: { userId_organizationId: { userId: req.user.userId, organizationId: body.organizationId } },
    });
    if (!membership || (membership.role !== 'CLUB_HEAD' && membership.role !== 'COMMITTEE_MEMBER')) {
      throw new ForbiddenException('Only Club Heads can post recruitment notices.');
    }
    return this.prisma.recruitmentCycle.create({
      data: { title: body.title, description: body.description, deadline: new Date(body.deadline), organizationId: body.organizationId }
    });
  }

  @UseGuards(JwtAuthGuard)
  @Post(':id/apply')
  async apply(@Param('id') id: string, @Req() req: any, @Body('pitch') pitch: string) {
    // FLAW FIX: Only Students can apply
    if (req.user.role !== 'STUDENT') {
      throw new ForbiddenException('Only students can apply for club recruitment.');
    }

    const cycle = await this.prisma.recruitmentCycle.findUnique({ where: { id } });
    if (!cycle) throw new BadRequestException('Cycle not found');

    if (new Date(cycle.deadline) < new Date()) {
      throw new BadRequestException('The deadline for this recruitment has passed.');
    }

    const existingMember = await this.prisma.membership.findUnique({
      where: { userId_organizationId: { userId: req.user.userId, organizationId: cycle.organizationId } }
    });
    if (existingMember) {
      throw new BadRequestException('You are already a member of this club.');
    }

    return this.prisma.application.create({
      data: { recruitmentCycleId: id, userId: req.user.userId, pitch },
      include: { recruitmentCycle: { include: { organization: { include: { memberships: { where: { role: 'CLUB_HEAD' } } } } } } }
    });
  }

  @UseGuards(JwtAuthGuard)
  @Get(':id/applications')
  getApplications(@Param('id') id: string) {
    return this.prisma.application.findMany({
      where: { recruitmentCycleId: id },
      include: { user: { include: { studentProfile: true } } }
    });
  }

  @UseGuards(JwtAuthGuard)
  @Patch('applications/:id')
  async updateAppStatus(@Param('id') id: string, @Body('status') status: string) {
    return this.prisma.application.update({ where: { id }, data: { status } });
  }
}

import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class SearchService {
  constructor(private prisma: PrismaService) {}

  async globalSearch(query: string) {
    if (!query || query.trim() === '') {
      return { users: [], organizations: [], events: [], opportunities: [] };
    }

    const searchString = query.trim();

    const [users, organizations, events, opportunities] = await Promise.all([
      // Search Users (Name or Email)
      this.prisma.user.findMany({
        where: {
          OR: [
            { email: { contains: searchString, mode: 'insensitive' } },
            { studentProfile: { OR: [{ firstName: { contains: searchString, mode: 'insensitive' } }, { lastName: { contains: searchString, mode: 'insensitive' } }] } },
            { facultyProfile: { OR: [{ firstName: { contains: searchString, mode: 'insensitive' } }, { lastName: { contains: searchString, mode: 'insensitive' } }] } },
          ],
        },
        include: { studentProfile: true, facultyProfile: true },
        take: 5,
      }),
      // Search Organizations
      this.prisma.organization.findMany({
        where: { name: { contains: searchString, mode: 'insensitive' }, status: 'ACTIVE' },
        take: 5,
      }),
      // Search Events
      this.prisma.event.findMany({
        where: { title: { contains: searchString, mode: 'insensitive' }, status: 'PUBLISHED' },
        take: 5,
      }),
      // Search Opportunities
      this.prisma.opportunity.findMany({
        where: { title: { contains: searchString, mode: 'insensitive' }, status: 'PUBLISHED' },
        take: 5,
      }),
    ]);

    // Format users so frontend doesn't need complex logic
    const formattedUsers = users.map(u => ({
      id: u.id,
      name: u.studentProfile ? `${u.studentProfile.firstName} ${u.studentProfile.lastName}` : u.facultyProfile ? `${u.facultyProfile.firstName} ${u.facultyProfile.lastName}` : u.email,
      email: u.email,
      role: u.globalRole,
      avatarUrl: u.avatarUrl,
    }));

    return { users: formattedUsers, organizations, events, opportunities };
  }
}

import { Controller, Get, Put, Post, Delete, Body, UseGuards, Req, Patch, Param, NotFoundException, UseInterceptors, UploadedFile, UnauthorizedException } from '@nestjs/common';
import { FileInterceptor } from '@nestjs/platform-express';
import { diskStorage } from 'multer';
import { extname } from 'path';
import * as argon2 from 'argon2';
import { PrismaService } from '../prisma/prisma.service';
import { JwtAuthGuard } from '../auth/jwt-auth.guard';
import { RolesGuard } from '../auth/roles.guard';
import { Roles } from '../auth/roles.decorator';
import { AuditService } from '../audit/audit.service';

@Controller('users')
export class UsersController {
  constructor(private prisma: PrismaService, private audit: AuditService) {}

  @UseGuards(JwtAuthGuard)
  @Get('profile')
  async getProfile(@Req() req: any) {
    const user = await this.prisma.user.findUnique({
      where: { id: req.user.userId },
      include: { studentProfile: true, facultyProfile: true }
    });
    if (!user) throw new NotFoundException('User not found');
    const { passwordHash, ...result } = user;
    return result;
  }

  @UseGuards(JwtAuthGuard)
  @Get(':id/public')
  async getPublicProfile(@Param('id') id: string) {
    const user = await this.prisma.user.findUnique({
      where: { id },
      include: { studentProfile: true, facultyProfile: true, memberships: { include: { organization: true } } }
    });
    if (!user) throw new NotFoundException('User not found');
    return { id: user.id, email: user.email, globalRole: user.globalRole, avatarUrl: user.avatarUrl, studentProfile: user.studentProfile, facultyProfile: user.facultyProfile, memberships: user.memberships };
  }

  @UseGuards(JwtAuthGuard)
  @Put('profile')
  async updateProfile(@Req() req: any, @Body() body: any) {
    if (req.user.role === 'STUDENT') {
      return this.prisma.studentProfile.update({
        where: { userId: req.user.userId },
        data: { bio: body.bio, skills: body.skills, projects: body.projects, interests: body.interests, githubUrl: body.githubUrl, linkedinUrl: body.linkedinUrl }
      });
    } else if (req.user.role === 'FACULTY') {
      return this.prisma.facultyProfile.update({ where: { userId: req.user.userId }, data: { bio: body.bio } });
    }
    return { message: 'Admin profile requires no bio updates.' };
  }

  @UseGuards(JwtAuthGuard)
  @Put('password')
  async changePassword(@Req() req: any, @Body() body: any) {
    const user = await this.prisma.user.findUnique({ where: { id: req.user.userId } });
    if (!user) throw new NotFoundException('User not found');
    
    const isValid = await argon2.verify(user.passwordHash, body.oldPassword);
    if (!isValid) throw new UnauthorizedException('Incorrect current password');
    
    const newHash = await argon2.hash(body.newPassword);
    await this.prisma.user.update({ where: { id: user.id }, data: { passwordHash: newHash } });
    return { message: 'Password updated successfully' };
  }

  @UseGuards(JwtAuthGuard)
  @Post('avatar')
  @UseInterceptors(FileInterceptor('file', {
    storage: diskStorage({
      destination: './uploads',
      filename: (req, file, cb) => {
        const randomName = Array(32).fill(null).map(() => (Math.round(Math.random() * 16)).toString(16)).join('');
        return cb(null, `${randomName}${extname(file.originalname)}`);
      }
    })
  }))
  async uploadAvatar(@Req() req: any, @UploadedFile() file: Express.Multer.File) {
    const avatarUrl = `http://localhost:3000/uploads/${file.filename}`;
    await this.prisma.user.update({ where: { id: req.user.userId }, data: { avatarUrl } });
    return { avatarUrl };
  }

  @UseGuards(JwtAuthGuard)
  @Delete('account')
  async deleteAccount(@Req() req: any) {
    const userId = req.user.userId;
    
    // Manually clean up relations to avoid foreign key constraint errors
    await this.prisma.application.deleteMany({ where: { userId } });
    await this.prisma.membership.deleteMany({ where: { userId } });
    await this.prisma.eventRegistration.deleteMany({ where: { userId } });
    await this.prisma.message.deleteMany({ where: { OR: [{ senderId: userId }, { receiverId: userId }] } });
    await this.prisma.notification.deleteMany({ where: { userId } });
    await this.prisma.exchangeItem.deleteMany({ where: { userId } });
    await this.prisma.announcement.deleteMany({ where: { organization: { memberships: { some: { userId, role: 'CLUB_HEAD' } } } } });
    
    // Delete profiles
    await this.prisma.studentProfile.deleteMany({ where: { userId } });
    await this.prisma.facultyProfile.deleteMany({ where: { userId } });
    
    // Finally, delete the user
    await this.prisma.user.delete({ where: { id: userId } });

    return { message: 'Account deleted successfully' };
  }

  @UseGuards(JwtAuthGuard, RolesGuard)
  @Roles('ADMIN')
  @Get('all')
  async getAllUsers() {
    return this.prisma.user.findMany({
      select: { id: true, email: true, globalRole: true, status: true, avatarUrl: true, createdAt: true, studentProfile: { select: { firstName: true, lastName: true } }, facultyProfile: { select: { firstName: true, lastName: true } } },
      orderBy: { createdAt: 'desc' }
    });
  }

  @UseGuards(JwtAuthGuard, RolesGuard)
  @Roles('ADMIN')
  @Patch(':id/status')
  async updateUserStatus(@Param('id') id: string, @Body('status') status: 'ACTIVE' | 'SUSPENDED', @Req() req: any) {
    const res = await this.prisma.user.update({ where: { id }, data: { status } });
    await this.audit.logAction(req.user.userId, 'CHANGE_USER_STATUS', id, `Changed status of ${res.email} to ${status}`);
    return res;
  }

  @UseGuards(JwtAuthGuard, RolesGuard)
  @Roles('ADMIN')
  @Get('faculty')
  async getFaculty() {
    return this.prisma.user.findMany({ where: { globalRole: 'FACULTY' }, include: { facultyProfile: true } });
  }
}
